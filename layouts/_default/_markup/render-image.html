{{- /* Markdown image render hook — responsive + lazy loading */ -}} {{- $src := .Destination -}} {{- $isRemote := or (hasPrefix $src "http://") (hasPrefix $src "https://") (hasPrefix $src "//") -}} {{- if and (not $isRemote) (hasPrefix $src "/") -}} {{- $src = absURL (strings.TrimPrefix "/" $src) -}} {{- end -}}
